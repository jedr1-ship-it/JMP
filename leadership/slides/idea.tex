%% 1. Question
\begin{frame}[plain]
\vspace{0.9cm}
\begin{center}
{\LARGE\textbf{How do school managers matter?}}
\end{center}
\vspace{0.7cm}
\begin{center}
\small
\begin{tabular}{@{}L{2.4cm}L{10.2cm}@{}}
\lab{Whether} & Does the managerial input matter once the selection of workers is shut down? \\[0.4em]
\lab{Through what} & Teacher effort and practice, classroom organization, parental inputs: the production process behind principal value added. \\[0.4em]
\lab{For whom} & Teacher wellbeing and family trust are welfare-relevant outcomes, not only mediators. \\[0.4em]
\lab{Who} & How dispersed are manager effects, and which pre-determined traits predict a good manager. \\
\end{tabular}
\end{center}
\end{frame}

%% 2. Two stories
\begin{frame}{Two stories, one design}
\vspace{0.15cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Selection} & Spain does not attract or select its best managers: no pay or career incentives, posts left vacant and filled by the administration, single candidates, committees that pass over the top-merit candidate. \\[0.5em]
\lab{Causal effect} & What a change of manager does to a school, identified from mandatory exits, in a system where the selection channel is shut and the incoming manager is not selected on quality. \\[0.5em]
\lab{Link} & The selection story describes who arrives and how; the causal story measures what the arrival does. The first is the interpretation of the second. \\[0.5em]
\lab{Division} & Two papers. Policy: deserted competitions, incentives, cross-region series (Montalb\'an). Identification: what a manager does when arriving by rule (this paper). \\
\end{tabular}
\end{frame}

%% 3. Motivating facts
\begin{frame}{Motivating facts: selection in Madrid, 2015--2018 \cit{Montalb\'an, 2026, correlational}}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Competition} & About 1,000 competitions: three in four with a single candidate, one in four contested, 4\% deserted (62 schools). In 39\% of contested cases the winner is not the top-merit candidate. \\[0.35em]
\lab{After a failure} & Achievement $-0.20$ SD, persistent; pupil, family, and teacher climate unchanged; the appointed principal reports lower satisfaction ($-0.26$) and more regret ($+0.20$). \\[0.35em]
\lab{Before a failure} & Not poorer or lower-achieving schools. Their principal already reports satisfaction $-0.66$ SD and autonomy problems $+0.86$ SD relative to other public schools. \\
\end{tabular}
\vspace{0.35cm}

\muted{\footnotesize Failed successions are preceded by a deteriorating principalship: selection on unobservables, observed. Mandatory exits remove it. The $-0.20$ is a school-level SD and appears in the competition year, before the appointee arrives; we treat it as an upper bound on selection, not as an effect.}
\end{frame}

%% 6. Setting
\begin{frame}{Setting}
\vspace{0.15cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Managers} & Principals are teachers selected for four-year terms, without authority over hiring, dismissal, pay, or curriculum. \\[0.35em]
\lab{Selection} & A per-school committee (three administration, two school) scores objective merits (0--15) and a project. Outcomes: contested, single candidate, or failed and filled by the administration. \\[0.35em]
\lab{Labor input} & Fixed across a change of principal. The selection channel is shut; what remains is the managerial input. \\[0.35em]
\lab{Mandatory exits} & Retirement at 65 and a two-renewal limit (twelve years), both pre-determined. \\
\end{tabular}
\end{frame}

\end{tikzpicture}

\vspace{-0.15cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Instrument} & A mandatory exit determined by age or tenure; $t=0$ is the first school year under the new principal (appointed 1 July). \\[0.3em]
\lab{Exclusion} & The exit date affects outcomes only through the change of principal. \\[0.3em]
\lab{Tests} & No pre-trends in any outcome. Retiring principals' schools balanced on pre-period characteristics: exits occur across the distribution, not in the worst schools. \\[0.3em]
\lab{Benchmark} & Voluntary and failed successions, estimated separately: the selection on unobservables shown on the facts slide. \\
\end{tabular}
\end{frame}

%% 8. Reduced form
\begin{frame}{Reduced form}
\vspace{0.15cm}
\small
$\tau_s$: year of the mandatory exit in school $s$. $i$: teacher, classroom, pupil, or parent.
\begin{equation*}
Y_{ist}=\alpha_s+\lambda_t+\sum_{k\neq -1}\beta_k\,\mathbf{1}[\,t-\tau_s=k\,]+\varepsilon_{ist}
\end{equation*}
\vspace{0.1cm}
\begin{tabular}{@{}L{2.6cm}L{10.4cm}@{}}
\lab{$\beta_{k\ge0}$} & Dynamic treatment effect $k$ years after the exit. \\[0.35em]
\lab{$\beta_{k<0}$} & Pre-trends. \\[0.35em]
\lab{Unit} & The school is the unit of treatment; $i$ are repeated cross-sections within it. Not-yet-treated controls; staggered-robust estimator; clustered by school. \\
\end{tabular}
\end{frame}

%% 9. LATE
\begin{frame}{Local average treatment effect}
\vspace{0.15cm}
\small
$Z_{st}=\mathbf{1}[t\ge\tau_s]$, $D_{st}=\mathbf{1}[\text{new principal}]$.
\begin{align*}
D_{st}&=\pi Z_{st}+\alpha_s+\lambda_t+u_{st} \\
Y_{ist}&=\beta\,\widehat{D}_{st}+\alpha_s+\lambda_t+\varepsilon_{ist}
\end{align*}
\vspace{0.1cm}
\begin{tabular}{@{}L{2.6cm}L{10.4cm}@{}}
\lab{Compliers} & At 65: a long-tenured incumbent replaced. At twelve years: a younger incumbent replaced. Two complier groups, one parameter to compare. \\[0.35em]
\lab{Decomposition} & $\beta=\theta_{\text{new}}-\theta_{\text{old}}$. The incumbent's pre-exit staff rating and merits separate losing a good manager from receiving a worse one. \\[0.35em]
\lab{Panels} & Same teacher (tutors re-observed) and same pupil (grade 3 to 6) where the data allow; to be verified. \\
\end{tabular}
\end{frame}

%% 10. Variance and heterogeneity
\begin{frame}{Variance of manager effects; heterogeneity}
\vspace{0.1cm}
\small
$Y_{st}=\mu_s+\theta_{p(s,t)}+e_{st}$. Around a mandatory exit, $\theta_{\text{old}}$ is replaced by $\theta_{\text{new}}$:
\begin{equation*}
\mathrm{Var}(\Delta Y_{st}\mid Z_{st}=1)-\mathrm{Var}(\Delta Y_{st}\mid Z_{st}=0)=2\sigma^2_{\theta}
\qquad\qquad
Y_{ist}=\ldots+\beta D_{st}+\delta\,D_{st}\times W_s+\varepsilon_{ist}
\end{equation*}
\vspace{0.05cm}
\begin{tabular}{@{}L{2.6cm}L{10.4cm}@{}}
\lab{$\sigma^2_\theta$} & Dispersion of manager effects without ranking managers \cit{Jones and Olken, 2005}. \\[0.3em]
\lab{$W_s$} & How the vacancy was filled (contested, single, failed); successor's merits and age relative to the incumbent; change in the principal's reported practices (LOMCE questionnaire); change in the school's educational project (text). \\[0.3em]
\lab{$\delta$} & Treatment effect heterogeneity by the dimensions of the change. \\
\end{tabular}
\end{frame}

%% 11. Extension
\begin{frame}{Extension: rules versus discretion in selecting managers}
\vspace{0.15cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Rule} & The merit score, observed for every admitted candidate. \\[0.4em]
\lab{Discretion} & The committee's choice; in 39\% of contested cases it overrides the merit ranking. \\[0.4em]
\lab{Instrument} & Administration members are named per committee (BOCM). If they sit on several committees and are not assigned on school characteristics, their propensity to override or to fail a competition instruments who is selected \cit{Hoffman, Kahn, and Li, 2018}. \\[0.4em]
\lab{Question} & Does discretion select better managers than the rule? \\
\end{tabular}
\end{frame}

%% 14. Risks
\begin{frame}{Risks, in order}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Power} & 100 to 200 mandatory exits among primary schools over three transitions; expected effects of 0.03--0.05 SD on scores, larger on survey indices; family means are precise. \\[0.3em]
\lab{Panels} & Share of tutors and pupils re-observed is unknown; within-unit designs are conditional on it. \\[0.3em]
\lab{Tenure limit} & Appointment histories since 2004 must be rebuilt from BOCM. \\[0.3em]
\lab{Compliance at 65} & Extensions to 70 exist; count principals above 65. \\[0.3em]
\lab{Timing} & $t=0$ as the first year under the successor, never the competition year. \\
\end{tabular}
