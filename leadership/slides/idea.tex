%% 1. Question
\begin{frame}[plain,c]
\vfill
\begin{center}
{\LARGE\textbf{What is the impact of a change}\\[0.25em]\textbf{in school leadership?}}
\end{center}
\vspace{0.5cm}
\begin{center}
{\large In a system where principals have limited authority\\[0.2em]and few teachers want the job.}
\end{center}
\vfill
\end{frame}

%% 2. Known / unknown
\begin{frame}{Related literature}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item One SD of principal quality moves achievement by 0.03 to 0.10 SD \cit{Branch, Hanushek, and Rivkin, 2012; Bartanen, Husain, and Liebowitz, 2024}.
\item A change of principal lowers achievement by 0.03 SD and raises teacher exit \cit{Bartanen, Grissom, and Rogers, 2019}.
\item The identified channel is personnel: who is hired, retained, and dismissed \cit{Grissom and Bartanen, 2019; Mu\~noz and Prem, 2024}.
\end{itemize}
\vspace{0.9cm}
What is not identified is the effect on the practice and effort of a fixed staff, on the organization of classrooms, and on parental inputs.
\end{frame}

%% 4. Setting
\begin{frame}{Institutional setting}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item Spanish principals are teachers elected for four years. They do not hire, dismiss, set pay, or set the curriculum.
\item A change of principal therefore changes leadership on a fixed staff.
\item Civil-servant teachers may retire at 60 and must retire at 65, with extensions possible.
\end{itemize}
\vspace{0.9cm}
Data: LOMCE standardized tests in mathematics, Spanish, and English, with questionnaires to parents, teachers, and principals, for the universe of Madrid public primary schools, 2015--16 to 2018--19.
\end{frame}

%% 6. Identification
\begin{frame}{Identification strategy}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item Instrument: the incumbent principal is aged 60 or older. The exit rate rises from about 6\% to 34\%.
\item Treated cohorts: schools whose principal changes in 2018 or in 2019. Controls: schools not yet treated.
\item Threat: retirement after 60 is voluntary, and may respond to how the school is doing.
\item Test: among principals aged 60 or older, schools whose principal leaves and schools whose principal stays show the same levels and trends before the exit.
\end{itemize}
\end{frame}

%% 7. Equation
\begin{frame}{Empirical specification}
\vspace{0.3cm}
Event study around the change of principal:
\begin{equation*}
Y_{ist}=\alpha_s+\lambda_t+\sum_{k\neq -1}\beta_k\,\mathbf{1}[\,t-\tau_s=k\,]+\varepsilon_{ist}
\end{equation*}
\vspace{0.1cm}
Instrumental variables, with the age of the incumbent as instrument:
\begin{equation*}
D_{st}=\pi\,\mathbf{1}[\text{age}_{s,t-1}\ge 60]+\alpha_s+\lambda_t+u_{st},\qquad
Y_{ist}=\beta\,\widehat{D}_{st}+\alpha_s+\lambda_t+\varepsilon_{ist}
\end{equation*}
\vspace{0.2cm}
{\small $\tau_s$: first school year under the new principal. $i$: a teacher, classroom, pupil, or parent. The school is the unit of treatment; standard errors clustered by school; staggered-robust estimator.}
\end{frame}

%% 8. Beyond the mean
\begin{frame}{Dispersion and heterogeneity of principal effects}
\vspace{0.2cm}
\noindent Dispersion of principal effects, from excess variance around retirement-driven exits \cit{Jones and Olken, 2005}:
\begin{equation*}
\mathrm{Var}(\Delta Y_{st}\mid \text{exit})-\mathrm{Var}(\Delta Y_{st}\mid \text{no exit})=2\sigma^2_{\theta}
\end{equation*}
\vspace{-0.1cm}
\par\noindent Heterogeneity by characteristics determined before the change:
\begin{equation*}
Y_{ist}=\alpha_s+\lambda_t+\beta D_{st}+\delta\,D_{st}\times W_s+\varepsilon_{ist}
\end{equation*}
\vspace{0.1cm}
{\small $W_s$ includes how the vacancy was filled, the successor's merits and age relative to the incumbent, the successor's practices as a teacher in the previous year, and the staff's rating of the incumbent.}
\end{frame}

%% 9. Risks
\begin{frame}{Challenges}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item Selection into retirement: the age instrument is valid only if leavers and stayers over 60 are balanced before the exit.
\item Power: about 100 retirement-driven changes over two cohorts. Enough for survey indices, tight for school-level test scores.
\item Short windows: one or two years before and after each change.
\item Measurement: changes are detected from the respondent's sex and year of birth; to be validated against appointment records.
\end{itemize}
\end{frame}

%% 10. Next
\begin{frame}{Next steps}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item Balance and pre-trends: leavers versus stayers among principals aged 60 or older.
\item Power: within-school variance of each outcome and minimum detectable effects with 100 treated schools.
\item Validation: changes in the questionnaires against named appointments in the competition records.
\end{itemize}
\vspace{0.6cm}
If the first test holds, the causal design stands. If not, the selection records support a descriptive paper on how principals are chosen.
\end{frame}
