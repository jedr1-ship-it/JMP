%% 1. Question
\begin{frame}[plain,c]
\vfill
\begin{center}
{\Large\textbf{Through which inputs do school principals}\\[0.3em]\textbf{affect learning, and how much do they matter?}}
\end{center}
\vfill
\end{frame}

%% 2. Known / unknown
\begin{frame}{Related literature}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item One SD of principal quality moves achievement by 0.03 to 0.10 SD \cit{Branch, Hanushek, and Rivkin, 2012; Bartanen, Husain, and Liebowitz, 2024}.
\item A change of principal lowers achievement by 0.03 SD and raises teacher exit \cit{Bartanen, Grissom, and Rogers, 2019}.
\item The identified channel is personnel: who is hired, retained, and dismissed \cit{Grissom and Bartanen, 2019; Mu\~noz and Prem, 2024}.
\end{itemize}
\vspace{0.9cm}
What is not identified is the effect on the practice and effort of a fixed staff, on the organization of classrooms, and on parental inputs.
\end{frame}

%% 4. Setting
\begin{frame}{Institutional setting}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item Spanish principals are teachers elected for four years. They do not hire, dismiss, set pay, or set the curriculum.
\item A change of principal therefore changes leadership on a fixed staff.
\item Terms end by rule: retirement at 65, and at most two renewals, twelve years.
\end{itemize}
\vspace{0.9cm}
Data: LOMCE standardized tests in mathematics, Spanish, and English for the universe of Madrid public primary schools, 2015--2018, with supplementary questionnaires to parents, teachers, and principals.
\end{frame}

%% 6. Identification
\begin{frame}{Identification strategy}
\vspace{0.3cm}
\begin{center}
\begin{tikzpicture}[x=0.9cm,y=1cm]
\draw[muted,line width=0.8pt] (0,0) -- (12.4,0);
\foreach \x/\lbl/\sub/\col in {0/Appointment/{year 0}/amber, 4.1/Renewal/{year 4}/teal, 8.2/Renewal/{year 8}/teal, 12.3/Mandatory exit/{year 12, or age 65}/amber}{
  \fill[\col] (\x,0) circle (0.17);
  \node[above=0.3cm,font=\small\bfseries] at (\x,0) {\lbl};
  \node[below=0.3cm,font=\footnotesize,text=muted] at (\x,0) {\sub};
}
\end{tikzpicture}
\end{center}
\vspace{0.5cm}
\begin{itemize}\setlength{\itemsep}{0.9em}
\item Treatment: an exit forced by age or tenure. Control: schools whose principal has not reached either limit.
\item Assumption: the exit date is unrelated to school shocks. Test: no pre-trends.
\item Voluntary and deserted successions estimated separately, as the selection benchmark.
\end{itemize}
\end{frame}

%% 7. Equation
\begin{frame}{Empirical specification}
\vspace{0.5cm}
\begin{equation*}
Y_{ist}=\alpha_s+\lambda_t+\sum_{k\neq -1}\beta_k\,\mathbf{1}[\,t-\tau_s=k\,]+\varepsilon_{ist}
\end{equation*}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{0.9em}
\item $\tau_s$: first school year under the new principal. $i$: a teacher, classroom, pupil, or parent.
\item The school is the unit of treatment; outcomes are repeated cross-sections within it.
\item Staggered-robust estimator, not-yet-treated controls, clustered by school. 2SLS on the realized change where the tenure rule does not bind.
\end{itemize}
\end{frame}

%% 8. Beyond the mean
\begin{frame}{Dispersion and heterogeneity of principal effects}
\vspace{0.2cm}
\noindent Dispersion of principal effects, from excess variance around rule-based exits \cit{Jones and Olken, 2005}:
\begin{equation*}
\mathrm{Var}(\Delta Y_{st}\mid \text{exit})-\mathrm{Var}(\Delta Y_{st}\mid \text{no exit})=2\sigma^2_{\theta}
\end{equation*}
\vspace{-0.1cm}
\par\noindent Heterogeneity by characteristics determined before the change:
\begin{equation*}
Y_{ist}=\alpha_s+\lambda_t+\beta D_{st}+\delta\,D_{st}\times W_s+\varepsilon_{ist}
\end{equation*}
\vspace{0.1cm}
{\small $W_s$ includes how the vacancy was filled, the successor's merits and age relative to the incumbent, the successor's practices as a teacher in the previous year, and the staff's rating of the incumbent.}
\end{frame}

%% 9. Risks
\begin{frame}{Challenges}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item Power: 100 to 200 mandatory exits in primary over three transitions; effects of 0.03--0.05 SD on scores, larger on survey indices.
\item Panels: the share of tutors and pupils re-observed is unknown.
\item The tenure rule needs appointment histories since 2004; the age rule needs the count of principals above 65.
\end{itemize}
\end{frame}

%% 10. Next
\begin{frame}{Next steps}
\vspace{0.4cm}
\begin{itemize}\setlength{\itemsep}{1.1em}
\item Count exits by age and by tenure among primary schools, by year.
\item Share of tutors re-observed across years.
\item Re-run the deserted-competition regression with mandatory exits as treatment and the incumbent's prior satisfaction as outcome: a flat pre-period is the pre-trend test.
\end{itemize}
\end{frame}
