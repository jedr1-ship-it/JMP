%% 1. The question
\begin{frame}[plain]
\vspace{1.3cm}
\begin{center}
{\LARGE\textbf{What does a school leader change,}\\[0.2em]\textbf{and how far does it reach?}}
\end{center}
\vspace{0.9cm}
\begin{center}
\begin{minipage}{0.78\textwidth}
Principals are an input in the production of education, and their value added is well documented. How leadership turns into teacher behavior, classroom organization, family engagement, and learning is not.
\end{minipage}
\end{center}
\end{frame}

%% 2. What we know
\begin{frame}{What we know}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Leaders matter} & Principal value added: 0.05 to 0.10 SD of achievement per SD of principal quality \cit{Branch, Hanushek, and Rivkin, 2012; B\"ohlmark, Gr\"onqvist, and Vlachos, 2016}; smaller once within-school variation is cleaned \cit{Bartanen, Husain, and Liebowitz, 2024}. \\[0.3em]
\lab{Changes matter} & Principal turnover lowers achievement by about 0.03 SD and raises teacher turnover \cit{Bartanen, Grissom, and Rogers, 2019}. \\[0.3em]
\lab{The channel so far} & Personnel: effective principals retain strong teachers and shed weak ones \cit{Grissom and Bartanen, 2019}; principals reshape the staff through hiring and retention \cit{Bartanen and Grissom, 2023}. \\[0.3em]
\lab{Not yet seen} & Under an exogenous change of leader: the same teachers' practice and wellbeing, the classrooms, and the families. \\
\end{tabular}
\end{frame}

%% 4. Intermediate outcomes
\begin{frame}{What a leader can change, and what we observe}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Teachers} & Classroom practice, collaboration, training, self-efficacy, wellbeing. \\[0.4em]
\lab{Classrooms} & Allocation of teachers and support hours across groups, discipline, safety. \\[0.4em]
\lab{Families} & Involvement, satisfaction with teachers and the school, perceived quality. \\[0.4em]
\lab{Pupils} & Engagement, perceptions of the school, achievement by subject. \\
\end{tabular}
\vspace{0.6cm}

Each is measured for every school and year. Teacher and family outcomes are dependent variables in their own right, and jointly the channels to achievement.
\end{frame}

%% 6. Setting
\begin{frame}{Setting}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Institutions} & Principals are teachers selected from the staff for four-year terms. Staffing, pay, and curriculum are set by the administration: a new principal changes leadership on a fixed staff. \\[0.4em]
\lab{Exogenous timing} & Terms end by compulsory retirement at 65 and a twelve-year limit, both fixed years in advance. \\[0.4em]
\lab{Successor} & Comes from the same staff; observed as a classroom teacher the year before appointment. \\[0.4em]
\lab{Data} & Universe of Madrid public primary schools, 2015 to 2018: principal and teacher questionnaires, timetables, all pupils and parents by classroom, standardized tests by subject. \\
\end{tabular}
\end{frame}

%% 9. Estimands
\begin{frame}{Estimands}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Average effect} & Of a mandatory leadership transition on each domain. \\[0.5em]
\lab{Variance} & Whether leaders differ: dispersion of outcome changes around rule-based transitions versus other years \cit{Jones and Olken, 2005}. No classification of leaders needed. \\[0.5em]
\lab{Heterogeneity} & By pre-determined characteristics of incoming and outgoing principals. \\
\end{tabular}
\vspace{0.6cm}

\muted{\footnotesize Whether leadership matters, how much leaders differ, and which observable traits predict effective ones.}
\end{frame}

%% 11. Open questions
\begin{frame}{Open questions}
\vspace{0.3cm}
\begin{itemize}\setlength{\itemsep}{0.9em}
\item \textbf{Mechanisms to prioritize.} Which intermediate outcomes should respond within the first year of a new principal, and which later?
\item \textbf{What the data must have.} A stable teacher identifier across years; the principal's age and years in office; the survey items that rate the principal.
\item \textbf{Sample of rule-based transitions} in 2015 to 2018, and access to appointment records and later years.
\end{itemize}
\end{frame}
