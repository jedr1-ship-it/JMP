%% 1. The question
\begin{frame}[plain]
\vspace{1.3cm}
\begin{center}
{\LARGE\textbf{What does a school leader change,}\\[0.2em]\textbf{and how far does it reach?}}
\end{center}
\vspace{0.9cm}
\begin{center}
\begin{minipage}{0.78\textwidth}
Principals are an input in the production of education, and their value added is well documented. How leadership turns into teacher behavior, classroom organization, family engagement, and learning is not.
\end{minipage}
\end{center}
\end{frame}

%% 2. What we know
\begin{frame}[label=know]{What we know}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Leaders matter} & Principal value added: 0.05 to 0.10 SD of achievement per SD of principal quality \cit{Branch, Hanushek, and Rivkin, 2012; B\"ohlmark, Gr\"onqvist, and Vlachos, 2016}; smaller once within-school variation is cleaned \cit{Bartanen, Husain, and Liebowitz, 2024}. \\[0.3em]
\lab{Changes matter} & Principal turnover lowers achievement by about 0.03 SD and raises teacher turnover \cit{Bartanen, Grissom, and Rogers, 2019}. \\[0.3em]
\lab{The channel so far} & Personnel: effective principals retain strong teachers and shed weak ones \cit{Grissom and Bartanen, 2019}; principals reshape the staff through hiring and retention \cit{Bartanen and Grissom, 2023}. \\[0.3em]
\lab{Not yet seen} & Under an exogenous change of leader: the same teachers' practice and wellbeing, the classrooms, and the families. \\
\end{tabular}
\vspace{0.1cm}
\hfill\hyperlink{recent}{\beamerbutton{Recent work}}
\end{frame}

%% 4. Intermediate outcomes
\begin{frame}{What a leader can change, and what we observe}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Teachers} & Classroom practice, collaboration, training, self-efficacy, wellbeing. \\[0.4em]
\lab{Classrooms} & Allocation of teachers and support hours across groups, discipline, safety. \\[0.4em]
\lab{Families} & Involvement, satisfaction with teachers and the school, perceived quality. \\[0.4em]
\lab{Pupils} & Engagement, perceptions of the school, achievement by subject. \\
\end{tabular}
\vspace{0.6cm}

Each is measured for every school and year. Teacher and family outcomes are dependent variables in their own right, and jointly the channels to achievement.
\end{frame}

%% 6. Setting
\begin{frame}{Setting}
\vspace{0.1cm}
\small
\begin{tabular}{@{}L{3.1cm}L{9.9cm}@{}}
\lab{Institutions} & Principals are teachers selected from the staff for four-year terms. Staffing, pay, and curriculum are set by the administration: a new principal changes leadership on a fixed staff. \\[0.4em]
\lab{Exogenous timing} & Terms end by compulsory retirement at 65 and a twelve-year limit, both fixed years in advance. \\[0.4em]
\lab{Successor} & Comes from the same staff; observed as a classroom teacher the year before appointment. \\[0.4em]
\lab{Data} & Universe of Madrid public primary schools, 2015 to 2018: principal and teacher questionnaires, timetables, all pupils and parents by classroom, standardized tests by subject. \\
\end{tabular}
\end{frame}

%% 7b. The clock in one equation
\begin{frame}{The clock, in one equation}
\vspace{0.1cm}
\small
Let $\tau_s$ be the year school $s$'s clock runs out: its principal turns 65, or completes year twelve.
\begin{equation*}
Y_{ist}=\alpha_s+\lambda_t+\sum_{k\neq -1}\beta_k\,\mathbf{1}[\,t-\tau_s=k\,]+\varepsilon_{ist}
\end{equation*}
\vspace{0.1cm}
\begin{tabular}{@{}L{2.6cm}L{10.4cm}@{}}
\lab{$i$} & A teacher, a classroom, a pupil, or a parent in school $s$ in year $t$. \\[0.3em]
\lab{$\beta_k,\ k\ge 0$} & The effect $k$ years after a rule-based change of principal, relative to the year before. \\[0.3em]
\lab{$\beta_k,\ k<0$} & Must be zero. The test that the change is not a response to the school. \\[0.3em]
\lab{Controls} & Schools whose clock has not yet run out. Clustered by school; staggered-robust weights. \\
\end{tabular}
\end{frame}

%% 7c. Effect of a change
\begin{frame}{Idea 1: the effect of a change of leader}
\vspace{0.1cm}
\small
$Z_{st}=\mathbf{1}[t\ge\tau_s]$ is the rule; $D_{st}=\mathbf{1}[\text{new principal in } s \text{ at } t]$ is what happened.
\begin{align*}
D_{st}&=\pi Z_{st}+\alpha_s+\lambda_t+u_{st} \\
Y_{ist}&=\beta\,\widehat{D}_{st}+\alpha_s+\lambda_t+\varepsilon_{ist}
\end{align*}
\vspace{0.1cm}
\begin{tabular}{@{}L{2.6cm}L{10.4cm}@{}}
\lab{$\beta$} & The effect of a new principal in schools where the rule forced the change. \\[0.3em]
\lab{$\pi$} & One at 65. Below one at twelve years: the principal may stand again and win. \\[0.3em]
\lab{Same people} & Replace $\alpha_s$ by $\alpha_j$ for the same teacher under two principals; add the grade-3 score for the same pupil in grade 6. \\
\end{tabular}
\end{frame}

%% 7d. Variance
\begin{frame}{Idea 2: do leaders differ at all?}
\vspace{0.1cm}
\small
If $Y_{st}=\mu_s+\theta_{p(s,t)}+e_{st}$, a rule-based transition replaces $\theta_{\text{old}}$ by $\theta_{\text{new}}$:
\begin{equation*}
\mathrm{Var}\big(\Delta Y_{st}\mid Z_{st}=1\big)-\mathrm{Var}\big(\Delta Y_{st}\mid Z_{st}=0\big)=2\,\sigma^2_{\theta}
\end{equation*}
\vspace{0.1cm}
\begin{tabular}{@{}L{2.6cm}L{10.4cm}@{}}
\lab{$\sigma^2_\theta$} & The variance of leader effects on each outcome: how much leaders differ, without ranking any of them \cit{Jones and Olken, 2005}. \\[0.3em]
\lab{Assumption} & $e_{st}$ is unrelated to the clock. The same pre-trend test as before. \\[0.3em]
\lab{Reading} & A zero mean effect with a large $\sigma^2_\theta$ means leaders matter and cancel out on average. \\
\end{tabular}
\end{frame}

%% 7e. Heterogeneity
\begin{frame}{Idea 3: what makes a good leader}
\vspace{0.1cm}
\small
\begin{equation*}
Y_{ist}=\alpha_s+\lambda_t+\beta\,D_{st}+\delta\,D_{st}\times W_s+\varepsilon_{ist}
\end{equation*}
\vspace{0.1cm}
\begin{tabular}{@{}L{2.6cm}L{10.4cm}@{}}
\lab{$W_s$} & Fixed before the new principal takes office: their own practices and wellbeing as a teacher in $\tau_s-1$; the committee's score of their project; the staff's rating of the outgoing principal in $\tau_s-1$. \\[0.3em]
\lab{$\delta$} & How the effect of the change varies with who leaves and who arrives. \\[0.3em]
\lab{Check} & $W_s$ does not predict $\beta_{k<0}$: the type of the successor is unrelated to where the school was going. \\
\end{tabular}
\end{frame}

%% 11. Open questions
\begin{frame}{Open questions}
\vspace{0.3cm}
\begin{itemize}\setlength{\itemsep}{0.9em}
\item \textbf{Mechanisms to prioritize.} Which intermediate outcomes should respond within the first year of a new principal, and which later?
\item \textbf{What the data must have.} A stable teacher identifier across years; the principal's age and years in office; the survey items that rate the principal.
\item \textbf{Sample of rule-based transitions} in 2015 to 2018, and access to appointment records and later years.
\end{itemize}
\end{frame}
