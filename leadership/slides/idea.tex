%% 1. The question
\begin{frame}[plain]
\vspace{1.2cm}
\begin{center}
{\LARGE\textbf{What can a school leader do}\\[0.2em]\textbf{when they cannot hire, fire, or pay?}}
\end{center}
\vspace{0.9cm}
\begin{center}
\begin{minipage}{0.78\textwidth}
Schools whose teachers value their principal show better climate, more collaboration, better classroom management, and higher teacher wellbeing \cit{OECD, TALIS 2018}. None of it is causal: good schools may simply make their principals look good.
\end{minipage}
\end{center}
\end{frame}

%% 2. Why Spain
\begin{frame}{Why Spain: the leader changes, the staff does not}
\vspace{0.3cm}
\begin{tabular}{@{}L{3.4cm}L{10.2cm}@{}}
\lab{The principal} & A teacher elected by colleagues for a four-year term, who returns to the classroom afterwards. The successor comes from the same staff. \\[0.3em]
\lab{Personnel power} & None. Teachers are civil servants assigned by the regional administration. The principal cannot hire, dismiss, set pay, or change the curriculum. \\[0.3em]
\lab{What is left} & The organization of the school: who teaches which class, how time and support hours are used, the plan and its priorities, discipline, the relation with families, in-school training. \\[0.3em]
\lab{Terms end by rule} & Compulsory retirement at 65, and at most two renewals: twelve years in office. Both dates are fixed years in advance, regardless of how the school is doing. \\
\end{tabular}
\vspace{0.5cm}

\muted{\small Evidence that principals matter comes from the United States, where the identified channel is the management of the teaching force \cit{Branch, Hanushek, and Rivkin, 2012; Grissom, Egalite, and Lindsay, 2021}. Here that channel is closed.}
\end{frame}

%% 5. What we test
\begin{frame}{What we would test, in the order effects should appear}
\vspace{0.2cm}
\begin{tabular}{@{}L{0.8cm}L{7.6cm}L{3.6cm}L{1.9cm}@{}}
\toprule
 & \textbf{Hypothesis} & \textbf{Measured with} & \textbf{When} \\
\midrule
\lab{H1} & The new leader reassigns teachers and support hours across classrooms & Schedules, class composition & Year 1 \\
\lab{H2} & Discipline and classroom climate change & Pupil and teacher reports & Years 1--2 \\
\lab{H3} & Family involvement and satisfaction change & Parent survey & Years 1--2 \\
\lab{H4} & Teacher self-efficacy and wellbeing change & Teacher survey & Years 1--2 \\
\lab{H5} & The same teachers change their classroom practice & Teacher survey, within teacher & Years 2--3 \\
\lab{H6} & Achievement moves, by subject & Standardized tests, pupil value added & Last \\
\bottomrule
\end{tabular}
\vspace{0.4cm}

\muted{\small Preregistered indices by domain; achievement is the last link, not the first. If rows 1 to 4 do not move, a result in row 6 would be hard to believe.}
\end{frame}

%% 6. Three parameters
\begin{frame}{Three parameters, not one}
\vspace{0.3cm}
\begin{tabular}{@{}L{3.4cm}L{10.2cm}@{}}
\lab{Mean} & The cost of forced rotation: what a change of leader that nobody asked for does to the school. Madrid imposes it with the twelve-year cap. \\[0.4em]
\lab{Variance} & Whether leaders matter at all: outcomes should disperse more after a rule-based exit than in an ordinary year if the identity of the leader matters \cit{Jones and Olken, 2005}. \\[0.4em]
\lab{Heterogeneity} & What makes a good one, with measures fixed before office: the successor's own practices as a teacher, the selection committee's grade of their project, the staff's rating of the outgoing principal. \\
\end{tabular}
\vspace{0.5cm}

\muted{\small The variance test needs no classification of who is good. The heterogeneity does, and uses only what was observed before the leader took office.}
\end{frame}

%% 7. Data
\begin{frame}{Data}
\vspace{0.2cm}
\begin{tabular}{@{}L{3.4cm}L{10.2cm}@{}}
\lab{In hand} & Universe of Madrid public primary schools, 2015 to 2018. Principal questionnaire. Teachers: practices, wellbeing, training. Schedules: who teaches which class. All pupils by classroom: scores by subject, perceptions. All parents by classroom: involvement, satisfaction. \\[0.4em]
\lab{To collect} & Principal appointments, renewals, and exits with cause (BOCM). The committee's grade of each project. Teacher assignment lists: chosen versus imposed posts. School plans from archived websites. Public school-level results from 2022: the long run. \\[0.4em]
\lab{Rotation} & 136 principals selected and 36 unplanned vacancies in 2016--17 across 1,156 public centres; 336 changes in 2017. Roughly 400 to 500 changes in primary over the three transitions, of which the rule-based subset is to be counted. \\
\end{tabular}
\end{frame}

%% 8. Open questions
\begin{frame}{Three things to decide}
\vspace{0.3cm}
\begin{itemize}\setlength{\itemsep}{0.9em}
\item \textbf{Which mechanisms go first.} Which levers and domains on the map do we believe a new principal actually moves in year one?
\item \textbf{What the data must have.} A stable teacher identifier across years; the principal's age and years in office; the survey items that rate the principal.
\item \textbf{How many rule-based exits there are} in 2015 to 2018, and whether the Consejer\'ia can add appointment records and later years.
\end{itemize}
\end{frame}
