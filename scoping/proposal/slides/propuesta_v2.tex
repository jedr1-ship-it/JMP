%% ---------- 1. Motivación ----------
\begin{frame}{Motivación}
\begin{itemize}
\item La primera infancia es un periodo sensible: el estrés del cuidador y la ruptura del entorno a los 0--2 años dejan huella en el desarrollo (Gunnar y Hostinar; Carneiro et al., \emph{EJ}).
\item Los conflictos civiles son el shock de estrés más extendido del mundo en desarrollo, y la literatura encuentra efectos de largo plazo de la exposición en la infancia (León 2012; Duque 2024; Akbulut-Yuksel, Tekin y Turan 2025).
\item Dos preguntas siguen abiertas:
\begin{enumerate}
\item ¿Importa la \textbf{edad} a la que llega la violencia, y por qué canal: la respuesta del hogar o la exposición directa del niño?
\item ¿Son creíbles los perfiles de edad existentes cuando la \textbf{respuesta dominante del hogar es irse}? La réplica de León (2012) con censos completos encuentra efectos antes de la concepción (Huaroto, Gonzales y Pacheco, \emph{JHR} 2026).
\end{enumerate}
\item Colombia 1997--2006 permite responder a ambas: una ola con actor, entrada escalonada por municipio, fin fechado, un registro de víctimas mensual y un registro de desplazados.
\end{itemize}
\end{frame}

%% ---------- 2. Este paper ----------
\begin{frame}{Este paper}
\textbf{Pregunta.} ¿Cómo afecta la edad a la que la violencia paramilitar alcanza a un niño nacido en un municipio afectado a su escolaridad y empleo adultos, y qué parte opera a través de la huida del hogar?
\vspace{0.6em}

\textbf{Tres contribuciones.}
\begin{enumerate}
\item \textbf{Medición.} Panel municipio$\times$mes$\times$perpetrador (CNMH) y de expulsados (RUV): entrada, pico y salida fechados para 451 municipios; \emph{dos relojes} de dosis, violencia y huida.
\item \textbf{Un resultado metodológico con evidencia.} Localizar al individuo donde se le observa de adulto produce perfiles de edad espurios: placebos pre-concepción distintos de cero en los dos relojes, tendencias monótonas sin salto en el evento y composición que se mueve en paralelo (3--8 M de observaciones).
\item \textbf{Efectos por edad por intención de tratar y descomposición de canales.} Municipio de nacimiento $\times$ cohorte (GEIH 2012--2024), dos relojes separables ($\rho = 0{,}5$), placebos y cotas por selección.
\end{enumerate}
\vspace{0.4em}
\muted{Las contribuciones 1 y 2 están hechas; la 3 es el núcleo de los próximos seis meses y tiene criterios de muerte explícitos.}
\end{frame}

%% ---------- 3. Avance de resultados ----------
\begin{frame}{Avance de resultados}
\small
\begin{enumerate}
\item \textbf{El calendario.} 94\,\% de las víctimas paramilitares 1997--2006 en 451 municipios; presencia mediana de 9 años; el 69--90\,\% de la caída de la violencia precede al acto de desmovilización: la variación útil es la \emph{entrada} y el \emph{pico}, no el acto.
\item \textbf{La huida.} Expulsión mediana del 23\,\% de la población en los municipios afectados; responde a la violencia total ($\rho = 0{,}63$), no al actor. La entrada paramilitar reduce un 18\,\% los evaluados de la cohorte en los municipios de entrada limpia.
\item \textbf{El artefacto.} Con la residencia a los 17 (Saber 11, 3--8 M), el perfil dosis-por-edad es positivo y creciente con la edad para cualquier perpetrador, el placebo pre-concepción es distinto de cero en los dos relojes y, alrededor de la salida o del pico, el puntaje es una pendiente monótona de $+0{,}23$ a $-0{,}11$ DE con la composición moviéndose en paralelo.
\item \textbf{Separabilidad.} La correlación residual entre la dosis de violencia y la de huida es 0,46--0,51 en todas las ventanas de edad: los dos canales se pueden estimar por separado.
\item \textbf{Lo que sigue.} Intención de tratar por municipio de nacimiento en la GEIH con los mismos dos relojes, placebos y cotas; go/no-go en el mes 6.
\end{enumerate}
\end{frame}

\end{tikzpicture}
\end{center}
\small
\begin{itemize}
\item El ``tratamiento'' que recibe un niño es un paquete: violencia en el municipio y respuesta del hogar. En esta ola la respuesta dominante es irse (una cuarta parte de la población).
\item Dos relojes medidos por separado: víctimas por 1.000 habitantes ($V$) y expulsados por 1.000 ($E$), ambos acumulados en ventanas de edad relativas al nacimiento.
\item Dos implicaciones: (i) la localización tiene que ser anterior al tratamiento, porque quien sigue en $m$ a los 17 no es quien estaba a los 2; (ii) con los dos relojes, ``hogar frente a exposición directa'' se convierte en una descomposición estimable.
\end{itemize}
\end{frame}

\begin{frame}{Predicciones como formas del perfil}
\begin{columns}[T]
\column{0.56\textwidth}
\includegraphics[width=\textwidth]{fig_h1h2_dos_relojes.pdf}
\column{0.42\textwidth}
\small
\textbf{H1$'$, canal hogar / huida.} El reloj de expulsiones manda: $|\delta_a|>0$, máximo en in utero y 0--2 y decreciente con la edad; $\beta_a \approx 0$ condicional a $E$.
\vspace{0.4em}

\textbf{H2$'$, exposición directa.} El reloj de víctimas manda y sólo desde que el niño puede codificar la experiencia: $\beta_a \neq 0$ para $a \ge 3$, condicional a $E$.
\vspace{0.4em}

\textbf{Mixta.} Ambos. \textbf{Condición de validez común:} las ventanas pre-concepción $[-45,-10]$ y $[-81,-46]$ meses valen cero en los dos relojes.
\vspace{0.4em}

\textbf{Condición de separabilidad:} correlación residual entre $V_a$ y $E_a$ por debajo de 0,7.
\end{columns}

%% ---------- 7. Estrategia empírica ----------
\begin{frame}{Estrategia empírica}
\small
Persona $i$ nacida en el municipio $m$ y la cohorte $c \in [1988, 2003]$, observada en el año $t$:
\begin{equation*}
Y_{imc} = \sum_a \beta_a\, V_{m,c+a} + \sum_a \delta_a\, E_{m,c+a} + \sum_a \varphi_a\, G_{m,c+a} + X_i'\gamma + \mu_m + \lambda_{c \times d(m)} + \theta_t + \varepsilon_{imc}
\end{equation*}
\begin{itemize}
\item Ventanas de longitud fija: pre-concepción (placebo), in utero, 0--2, 3--5, 6--11, 12--17. $G$: guerrilla y fuerza pública. $\mu_m$: municipio de nacimiento; $\lambda_{c\times d(m)}$: cohorte$\times$departamento; $\theta_t$: año de encuesta y edad al encuestar. Errores agrupados por $m$ (451 conglomerados).
\item \textbf{Qué identifica:} el calendario municipal de entrada y pico \emph{dentro} de cada departamento y cohorte: dos niños del mismo departamento y año de nacimiento cuyo municipio recibió la ola a edades distintas.
\item \textbf{Intención de tratar:} la dosis se asigna por municipio de nacimiento, no por dónde está la persona. La huida deja de ser un problema de muestra y pasa a ser un canal medido.
\item \textbf{Diseño complementario:} event study de cohortes relativas a la entrada y al pico municipal, con puntaje, número de nacidos y composición como resultados.
\item \textbf{Validez:} placebos pre-concepción en los dos relojes; nacimientos EEVV por $m\times c$ sobre la dosis; guerrilla como placebo de perpetrador; Rambachan-Roth; de Chaisemartin-D'Haultf\oe uille y Callaway-Goodman-Bacon-Sant'Anna para dosis continua; wild-cluster bootstrap.
\end{itemize}
\end{frame}

\begin{frame}{Lo que sigue: intención de tratar por municipio de nacimiento}
\small
\begin{columns}[T]
\column{0.52\textwidth}
\textbf{Datos.} GEIH 2012--2024: municipio de nacimiento, edad en años, escolaridad, ocupación, formalidad, ingreso; cohortes 1988--2003 observadas a los 20--36 años. EEVV 1998--2010 como denominador y composición al nacer.

\textbf{Estimación.} La ecuación de la estrategia empírica con $m$ = municipio de nacimiento y los dos relojes; mismas ventanas, placebos y controles por perpetrador. Primer ejercicio: el mismo perfil con municipio actual y con municipio de nacimiento, para las mismas cohortes. Una figura, dos perfiles: el artefacto medido dentro de la misma encuesta.
\vspace{0.4em}

\textbf{Outcomes.} Años de escolaridad, secundaria completa, educación superior, empleo, formalidad, ingreso laboral.
\column{0.46\textwidth}
\textbf{Potencia.} La restricción son los 451 conglomerados, no $N$; se calcula con los conteos por municipio de nacimiento $\times$ cohorte en la primera semana. Criterio: MDE $\le 0{,}05$ DE en escolaridad para un contraste binario por tramo.
\vspace{0.4em}

\textbf{Amenazas medidas, no asumidas.} Municipio de nacimiento autodeclarado (cociente ocurrencia/residencia en EEVV); emigración exterior y mortalidad violenta (defunciones, cotas); composición endógena de nacimientos (nacimientos por $m\times c$ sobre la dosis); encuesta repetida frente a registro (potencia).
\end{columns}
\end{frame}

%% ---------- 11. Conclusión ----------
\begin{frame}{Conclusión}
\begin{itemize}
\item Una ola de violencia con actor, entrada escalonada y fin fechado, medida mes a mes, y un registro de la respuesta de los hogares: dos relojes con los que la pregunta de edad de exposición se puede escribir como una descomposición de canales.
\item Resultado ya establecido: cuando el hogar responde yéndose, localizar al niño donde se le observa de adulto produce perfiles de edad espurios. Lo mostramos con 3--8 millones de observaciones, placebos en los dos relojes y event studies en entrada, pico y salida.
\item Los dos relojes son separables, de modo que el test hogar frente a exposición directa es estimable con localización al nacer.
\item Lo que se entrega en seis meses: efectos por edad por intención de tratar en la GEIH con placebos y cotas, o la evidencia de que la pregunta no es contestable con datos públicos.
\end{itemize}
