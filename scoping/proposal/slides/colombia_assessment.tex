%% 1
\begin{frame}{Question}
\lead{Does the long-run effect of a violent shock depend on the age at which the child receives it: before birth, in infancy, in preschool, at school age?}
\begin{itemize}
\item Sensitive periods: the same shock should leave different scars at different ages (Cunha and Heckman, 2007, \emph{AER}; Almond and Currie, 2011, \emph{JEP}).
\item For violence, the credible evidence stops at birth: the full-census replication of León (2012, \emph{JHR}) keeps the in-utero effect and rejects the postnatal profile, because the arrival of violence tracks pre-existing trends (Huaroto, Gonzales and Pacheco, 2026, \emph{JHR}).
\item So the age profile after birth is, today, unidentified.
\end{itemize}
\end{frame}

%% 2
\begin{frame}{Setting: the end of violence instead of its arrival}
\begin{columns}[T]
\column{0.52\textwidth}
\includegraphics[width=\textwidth]{fig1_en.pdf}
\column{0.46\textwidth}
\small
\begin{itemize}
\item Paramilitary blocs demobilize one by one, 2003--2006; zero victims after 2007. The end month $T_m$ is observed for each municipality.
\item Within a municipality, age at $T_m$ traces the whole profile: born after (never exposed), just before (in utero only), 1--2 years before (infancy), 3--5 (preschool), 6--11 (school age).
\item The end date comes from a national negotiation, not from local trends.
\end{itemize}
\end{columns}
\end{frame}

%% 3
\begin{frame}{Specification}
$i$ born in municipality $m$, cohort $c$; $k = c - T_m$ is age at the end.
\begin{equation*}
Y_{imc} = \sum_{k \ne -1} \beta_k\, \mathbf{1}[c - T_m = k] + \mu_m + \lambda_{c \times d(m)} + \varepsilon_{imc}
\end{equation*}
\begin{itemize}
\item $\beta_k$ is the effect of exposure up to age $k$, relative to the never-exposed cohort of the same municipality, net of department-by-cohort trends. The profile $\beta_0, \beta_1, \ldots, \beta_{11}$ is the object.
\item Staggered timing: Callaway and Sant'Anna (2021); de Chaisemartin and D'Haultf\oe uille (2024). Validity: pre-conception placebos; Rambachan and Roth (2023) bounds.
\item Municipality of birth, individual microdata: GEIH at 20--36 (schooling, employment, earnings); Saber 11 at 17.
\end{itemize}
\end{frame}

%% 4
\begin{frame}{What it would add}
\begin{itemize}
\item \textbf{Identification, not question.} Every age-profile paper uses the arrival or intensity of violence; the replication shows that variation is not exogenous. The end of violence is a different source. If it passes the placebo, this is the first credibly identified postnatal profile.
\item \textbf{Prenatal versus postnatal} in one design: $\beta_0$ against $\beta_1, \beta_2$.
\item \textbf{Mechanism:} profile by household displacement (migration module) and by maternal health at birth (vital statistics).
\end{itemize}
\vspace{0.6em}
\muted{The age-profile question itself is not new. The contribution is the source of variation.}
\end{frame}

%% 5
\begin{frame}{Assessment}
\textbf{Already well covered.} Bundervoet, Verwimp and Akresh (2009, \emph{JHR}); Akresh, Bhalotra, Leone and Osili (2012, \emph{AER P\&P}); León (2012, \emph{JHR}); Duque (2024, \emph{JCR}); Akbulut-Yuksel, Tekin and Turan (2025, \emph{JHR}). Same question, same fixed-effects design.
\vspace{0.7em}

\textbf{Too little variation.} 70\% of end dates fall within eighteen months. In-utero-only is a nine-month birth window; never-exposed cohorts become adults in 2026; with current data they exist in 28 municipalities.
\vspace{0.7em}

\textbf{Verdict.} The only novelty is the instrument, and the instrument is thin. Not a job market paper as it stands.
\end{frame}
