%% 1
\begin{frame}{Question}
\lead{Does the long-run effect of a shock depend on the age at which the child receives it: before birth, in infancy, in preschool, at school age?}
\begin{itemize}
\item Skill formation has sensitive periods, so the same shock should matter differently by age (Cunha and Heckman, 2007, \emph{AER}; Almond, Currie and Duque, 2018, \emph{JEL}).
\item Direct comparisons of one shock across ages are rare. For violence they exist (appendix) and share one design: cohorts of the same birthplace hit at different ages, with variation from the arrival or intensity of violence.
\item Two weaknesses of that design: where and when violence arrives follows local conditions, and how long a child is exposed depends on whether the household leaves. Both are documented, not assumed: the full-census replication of León (2012, \emph{JHR}) finds pre-conception effects, and León himself cannot observe migration histories.
\end{itemize}
\end{frame}

\begin{frame}{Specification}
$i$ born in municipality $m$, cohort $c$; $k = c - T_m$ is age at the end.
\begin{equation*}
Y_{imc} = \sum_{k \ne -1} \beta_k\, \mathbf{1}[c - T_m = k] + \mu_m + \lambda_{c \times d(m)} + \varepsilon_{imc}
\end{equation*}
\begin{itemize}
\item $\beta_k$ is the effect of exposure up to age $k$, relative to the never-exposed cohort of the same municipality, net of department-by-cohort trends. The profile $\beta_0, \beta_1, \ldots, \beta_{11}$ is the object.
\item Staggered timing: Callaway and Sant'Anna (2021); de Chaisemartin and D'Haultf\oe uille (2024). Validity: pre-conception placebos; Rambachan and Roth (2023) bounds.
\item Municipality of birth, individual microdata: GEIH at 20--36 (schooling, employment, earnings); Saber 11 at 17.
\end{itemize}
\end{frame}

%% 4
\begin{frame}{What it would add}
\begin{itemize}
\item \textbf{A different source of variation.} Age at the \emph{end} of a shock, with the end set by a national negotiation. It addresses the two weaknesses at once: timing not driven by local trends, location fixed at birth.
\item \textbf{The whole profile in one design:} prenatal only, infancy, preschool, school age, against never-exposed cohorts of the same municipality.
\item \textbf{Mechanism:} profile by household displacement (migration module) and by maternal health at birth (vital statistics).
\end{itemize}
\vspace{0.6em}
\muted{The question is not new. What would be new is a credibly identified postnatal profile.}
\end{frame}

%% 5
\begin{frame}{Assessment}
\textbf{Already well covered.} Bundervoet, Verwimp and Akresh (2009, \emph{JHR}); Akresh, Bhalotra, Leone and Osili (2012, \emph{AER P\&P}); León (2012, \emph{JHR}); Duque (2024, \emph{JCR}); Akbulut-Yuksel, Tekin and Turan (2025, \emph{JHR}). Same question, same fixed-effects design.
\vspace{0.7em}

\textbf{Too little variation.} 70\% of end dates fall within eighteen months. In-utero-only is a nine-month birth window; never-exposed cohorts become adults in 2026; with current data they exist in 28 municipalities.
\vspace{0.7em}

\textbf{Verdict.} The only novelty is the instrument, and the instrument is thin. Not a job market paper as it stands.
\end{frame}
