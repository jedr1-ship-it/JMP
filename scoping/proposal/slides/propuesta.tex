%% 1
\begin{frame}{What is the question?}
\keyfact{Early-life shocks have different effects depending on the age at which they hit. Does this hold for violence, and through which channel?}
\begin{itemize}
\item Sensitive periods in skill formation: Cunha and Heckman (2007, \emph{AER}); Almond and Currie (2011, \emph{JEP}); Carneiro, Salvanes, Willage and Willén (forthcoming, \emph{EJ}).
\item Childhood exposure to violence lowers adult human capital: León (2012, \emph{JHR}); Duque (2024, \emph{JCR}); Akbulut-Yuksel, Tekin and Turan (2025, \emph{JHR}).
\item What is not known: the \textbf{age gradient}. Is exposure at 0--2 worse than at 3--5 or at school age? Does the damage run through the caregiver or through the child's own experience?
\item Why it is hard: it requires a shock that \emph{starts and ends} on known dates, and children located \emph{before} the shock. Without both, the profile is contaminated: the replication of León by Huaroto, Gonzales and Pacheco (2026, \emph{JHR}) finds ``effects'' of violence that occurred before conception.
\end{itemize}
\end{frame}

\begin{itemize}

%% 4
\begin{frame}{What is the challenge? Households move}
\keyfact{Colombia has one of the largest conflict-driven internal migrations in the world: in affected municipalities the median share of population displaced in 1997--2010 is 23\%.}
\begin{itemize}
\item Exposure depends on whether the household stays and when it leaves; leaving is itself a different treatment.
\item Whoever is observed in the municipality years later is not who was there: locating people at the outcome mixes the effect with selection.
\end{itemize}
\vspace{0.6em}
\textbf{Our answer.}
\begin{itemize}
\item Locate children \textbf{before} the shock: exposure is assigned by \emph{municipality of birth}, in individual microdata. The estimand is intention to treat: the effect of being born where violence will end at age $a$, household response included.
\item Displacement is measured, not assumed: date and reason of the last move (GEIH) and displaced population by municipality and year (RUV).
\end{itemize}
\end{frame}

%% 5
\begin{frame}{How do we identify the effect?}
Person $i$ born in municipality $m$ in year $c$, observed in survey year $t$:
\begin{equation*}
Y_{imct} = \sum_{a} \beta_a\, D^{a}_{mc} + \gamma\, \text{Post}_{mc} + \varphi\, G_{mc} + \mu_m + \lambda_{c \times d(m)} + \theta_{t} + \varepsilon_{imct}
\end{equation*}
\small
\begin{itemize}
\item $a \in \{$in utero, 0--2, 3--5, 6--11$\}$. $D^{a}_{mc}$: years of paramilitary presence in $m$ (between $S_m$ and $T_m$) that cohort $c$ lived through at age $a$; zero in untreated municipalities and at ages after $T_m$. $\text{Post}_{mc}$: born after $T_m$ in a treated municipality.
\item $G_{mc}$: years of guerrilla presence at the same ages. $\mu_m$: municipality of birth. $\lambda_{c\times d(m)}$: cohort $\times$ department. $\theta_t$: survey year and age at survey. Standard errors clustered by municipality of birth (451 treated, 670 comparison).
\item \textbf{$\beta_a$ is the effect of one year of paramilitary violence lived at age $a$}, comparing cohorts born in the same municipality who were of different ages when the violence ended, net of department-level cohort trends. The variation comes from the timing of $T_m$ (and $S_m$) within departments.
\item Outcomes: years of schooling, secondary completion, tertiary enrollment, employment, formality, labor earnings.
\end{itemize}
\end{frame}

\begin{frame}{What do we expect, and what would it tell us?}
\begin{center}
\includegraphics[width=0.88\textwidth]{fig_h1h2_en.pdf}
\end{center}
\small
\begin{itemize}
\item \textbf{H1, household channel:} the damage runs through the caregiver (stress, displacement, income loss): it starts in utero, peaks at 0--2, and those born \emph{after} the end inherit part of it ($\gamma<0$).
\item \textbf{H2, direct exposure:} the child must be old enough to encode the experience: nothing before age 3, rising damage with age, nothing for those born after the end ($\gamma = 0$).
\item Second step: $\beta_a$ separately for households that were displaced and those that stayed.
\end{itemize}
\end{frame}

\begin{frame}{What is the contribution?}
\begin{enumerate}
\item \textbf{The age gradient of a violence shock, from an episode that ends.} First estimates separating in utero, 0--2, 3--5 and 6--11, with the end of exposure dated by municipality and location fixed before the shock. The dated end is what makes the comparison clean: cohorts born in the same municipality before and after the violence stopped.
\item \textbf{A mechanism test:} household channel versus direct exposure, from the shape of the profile and from those born after the end; then by household displacement.
\item \textbf{Adult outcomes:} schooling and labor market at ages 20--36, not only school test scores.
\end{enumerate}
\vspace{0.6em}
\small
Relative to Duque (2024, \emph{JCR}): a different decade, an actor with a dated end, municipality of birth instead of residence, adult outcomes. Relative to León (2012, \emph{JHR}) and its replication: the pre-conception placebo is part of the design, not an afterthought.
\vspace{0.6em}

\textbf{Timeline.} Months 1--2: data and power. Months 3--5: main estimates, placebos, displacement split. Month 6: first draft and go/no-go. Months 7--18: robustness and full paper.
\end{frame}
